def _run_branches(context, runtime, replay_records, *, resume):
    _require_writer()

    validated = _reconcile_durable_ledger(runtime, context)

    manifest = json.loads(manifest_path.read_text())
    manifest.update(_execution_audit(runtime, validated, context))
    models = None
    started = time.monotonic()
    try:
        if len(validated) < 60:
            models = _load_simultaneous(ADAPTER)
            if models["fallback_used"] is not False or models["recovery_backend"] is getattr(models["sft_policy"], "_backend", models["sft_policy"]):
                raise RuntimeError("R18-C2 requires distinct frozen recovery/SFT models without fallback")
            manifest.update(model_load_seconds=models["load_seconds"], gpu_memory=models["gpu_memory"], fallback_used=models["fallback_used"])
            write_json(manifest_path, manifest)
        for position, prefix in enumerate(context["prefixes"], start=1):
            pid = prefix["prefix_id"]
            if pid in validated:
                continue
            events_path = runtime / "execution_events.jsonl"
            past = load_jsonl(events_path) if events_path.exists() else []
            attempt = 1 + sum(e["prefix_id"] == pid and e["event"] == "started" for e in past)
            append_jsonl(events_path, {"prefix_id": pid, "event": "started", "attempt": attempt})
            compact, raw = _run_branch(context, prefix, replay_records[position - 1], models["recovery_backend"], models["sft_policy"])
            _validate_branch_record(raw, prefix, context)
            write_json(runtime / "branches" / f"{pid}.json", raw)
            append_jsonl(runtime / "card_v2_branches.jsonl", raw)

            validated[pid] = raw
